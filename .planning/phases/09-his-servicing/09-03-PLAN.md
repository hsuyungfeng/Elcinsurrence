```python
class CaseStore:
    def __init__(self, db_path: str | None = None): ...  # None 時用 config.settings.CASES_DB_PATH

    def create(self, *, case_id: str, kind: str, case_seq: str | None = None,
               order_code: str | None = None, payload: dict | None = None,
               actor: str | None = None) -> CaseRecord: ...
        # 拋 DuplicateCaseError（同 case_id 已存在）、
        # UnsafeIdentifierError（case_id 含路徑成分）、
        # CaseStoreError（kind 不是 "sampling"/"appeal"）

    def get(self, case_id: str) -> CaseRecord: ...  # 拋 CaseNotFoundError

    def transition(self, case_id: str, to_state: str, *, reason: str | None = None,
                    actor: str | None = None) -> CaseRecord: ...
        # 拋 CaseNotFoundError / IllegalTransitionError / MissingFailureReasonError

    def list_all(self, *, kind: str | None = None, limit: int = 1000) -> tuple[CaseRecord, ...]: ...
    def list_by_state(self, state: str, *, kind: str | None = None, limit: int = 100) -> tuple[CaseRecord, ...]: ...
    def counts_by_state(self, *, kind: str | None = None) -> dict[str, int]: ...

# 例外（自 elc_audit_engine.case_store re-export）：
# CaseStoreError, DuplicateCaseError, CaseNotFoundError, MissingFailureReasonError
# IllegalTransitionError, UnknownStateError

# CaseRecord 欄位：case_id/kind/state/case_seq/order_code/payload(dict|None)/
#                  created_at/updated_at/failure_reason
```
